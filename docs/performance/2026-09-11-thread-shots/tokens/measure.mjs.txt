const probe = async (spec) => {
  const res = {};
  for (const [key, [sel, props, pseudo]] of Object.entries(spec)) {
    res[key] = await page.evaluate(([sel, props, pseudo]) => {
      const el = document.querySelector(sel);
      if (!el) return null;
      const cs = getComputedStyle(el, pseudo ?? undefined);
      const o = {};
      for (const p of props) o[p] = cs.getPropertyValue(p);
      const r = el.getBoundingClientRect();
      o["#box"] = `${Math.round(r.width * 100) / 100}x${Math.round(r.height * 100) / 100}`;
      return o;
    }, [sel, props, pseudo ?? null]);
  }
  return res;
};

const hoverProbe = async (hoverSel, readSel, props) => {
  const h = page.locator(hoverSel).first();
  if (!(await h.count())) return null;
  const read = async () => page.evaluate(([s, p]) => { const el = document.querySelector(s); if (!el) return null; const cs = getComputedStyle(el); const o = {}; for (const q of p) o[q] = cs.getPropertyValue(q); return o; }, [readSel, props]);
  const idle = await read();
  await h.hover(); await page.waitForTimeout(350);
  const hov = await read();
  await page.mouse.move(5, 5); await page.waitForTimeout(250);
  return { idle, hover: hov };
};
