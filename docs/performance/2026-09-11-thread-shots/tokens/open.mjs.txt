export async function openApp(p, url) {
  await p.goto(url, { waitUntil: "domcontentloaded" });
  for (let i = 0; i < 80; i++) {
    await p.waitForTimeout(400);
    if (await p.locator('.work-block').count()) { await p.waitForTimeout(1200); return true; }
    const cont = p.locator('button.welcome-hint:not([disabled])');
    if (await cont.count()) { await cont.click().catch(() => {}); continue; }
    const name = p.locator('#welcome-name');
    if (await name.count()) {
      await name.fill("Stephen");
      await p.locator('button.welcome-continue:not([disabled])').first().click().catch(() => {});
      continue;
    }
    const session = p.getByText("Session s-0001", { exact: true }).first();
    if (await session.count()) await session.click().catch(() => {});
  }
  return false;
}
