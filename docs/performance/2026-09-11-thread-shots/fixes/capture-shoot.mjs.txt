import { chromium } from "/Users/stephen/.npm/_npx/705bc6b22212b352/node_modules/playwright/index.mjs";
import { mkdirSync, writeFileSync } from "node:fs";
import { openApp } from "./open.mjs";

const out = process.argv[2];
const url = "http://localhost:1420/?rps=0&approvals=manual";
mkdirSync(out, { recursive: true });
const log = [];
const say = (m) => { log.push(m); console.log(m); };

const browser = await chromium.launch();
const page = await browser.newPage({
  viewport: { width: 1440, height: 1000 },
  deviceScaleFactor: 2,
  colorScheme: "dark",
});
const errors = [];
page.on("pageerror", (e) => errors.push(String(e)));
say(`opened ${await openApp(page, url)}`);
await page.waitForTimeout(2000);

const shot = async (name, locator) => {
  const target = locator ?? page;
  try { await target.screenshot({ path: `${out}/${name}.png` }); say(`shot ${name}`); }
  catch (e) { say(`shot ${name} FAILED: ${e.message}`); }
};

await shot("01-full");

// D1 - vendored brand text anywhere in the rendered document
say(`D1 brand-in-dom ${JSON.stringify(await page.evaluate(() => {
  const t = document.body.innerText;
  return { chatgpt: t.includes("ChatGPT"), openai: /OpenAI/.test(t) };
}))}`);

// D2 - the turn header's data
say(`D2 turn-header ${JSON.stringify(await page.locator(".thread-turn-duration").first().textContent().catch(() => null))}`);
const headerRow = page.locator(".work-block .thread-activity-timeline__toggle-row").first();
await shot("02-turn-header", headerRow);

// D6 - hover is text-only, no fill
const toggle = page.locator(".work-block .thread-activity-timeline__toggle").first();
const idle = await toggle.evaluate((el) => [getComputedStyle(el).backgroundColor, getComputedStyle(el).color]);
await toggle.hover();
await page.waitForTimeout(300);
const hover = await toggle.evaluate((el) => [getComputedStyle(el).backgroundColor, getComputedStyle(el).color]);
say(`D6 turn-header idle=${JSON.stringify(idle)} hover=${JSON.stringify(hover)}`);
await shot("03-turn-header-hover", headerRow);

// D3 - approval card summary, and whether it clips
const card = page.locator(".thread-approval-request").first();
await shot("04-approval-card", card);
say(`D3 approval-body ${JSON.stringify(await card.locator(".approval-body").first().evaluate((el) => ({ scrollWidth: el.scrollWidth, clientWidth: el.clientWidth })).catch(() => null))}`);
say(`D3 approval-card ${JSON.stringify(await card.evaluate((el) => ({ scrollWidth: el.scrollWidth, clientWidth: el.clientWidth })))}`);
say(`D3 approval-text ${JSON.stringify((await card.innerText()).slice(0, 700))}`);
const raw = card.locator("details.approval-raw").first();
if (await raw.count()) {
  await raw.locator("summary").click();
  await page.waitForTimeout(250);
  await shot("05-approval-raw-open", card);
  say(`D3 raw-visible ${await card.locator("details.approval-raw pre.excerpt").first().isVisible()}`);
} else say("D3 raw-disclosure ABSENT");

// D8 - the composer strip while the card is on screen
say(`D8 strip-with-card-on-screen ${await page.locator(".composer-approval-strip").count()}`);

// D5 / D7 - open the activity fold, the command group, then every command row
const fold = page.locator(".work-block .thread-activity-timeline__toggle").first();
if ((await fold.getAttribute("aria-expanded")) === "false") { await fold.click(); await page.waitForTimeout(500); }
const group = page.locator(".work-activity-row button", { hasText: "Ran commands" }).first();
if (await group.count()) { await group.click(); await page.waitForTimeout(500); }
for (const row of await page.locator('.work-group-details .work-activity-row button[aria-expanded="false"]').all()) {
  await row.click().catch(() => {});
  await page.waitForTimeout(200);
}
await page.waitForTimeout(400);
await shot("06-activity-expanded");

const reasoning = page.locator('[data-slot="reasoning-panel"] button').first();
if (await reasoning.count()) {
  say(`D5 reasoning-label ${JSON.stringify((await reasoning.innerText()).trim())}`);
  await shot("07-reasoning", page.locator('[data-slot="reasoning-panel"]').first());
}

const footers = await page.locator(".thread-command-execution__footer").evaluateAll((els) =>
  els.map((el) => ({
    status: el.dataset.status,
    text: el.innerText.trim(),
    color: getComputedStyle(el).color,
    marker: getComputedStyle(el, "::before").backgroundColor,
  })),
);
say(`D7 command-footers ${JSON.stringify(footers)}`);
const failed = page.locator('.thread-command-execution:has(.thread-command-execution__footer[data-status="failed"])').first();
if (await failed.count()) await shot("08-command-failed", failed);
const ok = page.locator('.thread-command-execution:has(.thread-command-execution__footer[data-status="completed"])').first();
if (await ok.count()) await shot("09-command-ok", ok);

// D8 - a short window puts the card out of view; the strip should come back
await page.setViewportSize({ width: 1000, height: 520 });
await page.waitForTimeout(800);
await page.evaluate(() => {
  const el = [...document.querySelectorAll("*")].find((e) => e.scrollHeight > e.clientHeight + 40 && e.clientHeight > 150);
  if (el) el.scrollTop = 0;
});
await page.waitForTimeout(1200);
const inView = await page.locator(".thread-approval-request").first().evaluate((el) => {
  const r = el.getBoundingClientRect();
  return r.bottom > 0 && r.top < window.innerHeight;
}).catch(() => null);
say(`D8 short-window card-in-viewport=${inView} strip=${await page.locator(".composer-approval-strip").count()}`);
await shot("10-short-window");

say(`pageerrors ${JSON.stringify(errors)}`);
writeFileSync(`${out}/observations.txt`, log.join("\n") + "\n");
await browser.close();
