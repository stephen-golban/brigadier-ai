import { chromium } from "/Users/stephen/.npm/_npx/705bc6b22212b352/node_modules/playwright/index.mjs";
import { openApp } from "./open.mjs";
import { mkdirSync, writeFileSync } from "node:fs";
const out = process.argv[2]; mkdirSync(out,{recursive:true});
const log=[]; const say=m=>{log.push(m);console.log(m);};
const b = await chromium.launch();

async function peekWithApproval(page) {
  const tog = page.getByRole("button", { name: "Toggle Sidebar" }).first();
  await tog.click(); await page.waitForTimeout(900);
  await page.mouse.move(700,700); await page.waitForTimeout(400);
  await tog.hover(); await page.waitForTimeout(1800);
  // Into the previewed panel: the peek survives, and the toggle's tooltip (which would
  // otherwise swallow Escape before either handler sees it) closes.
  await page.mouse.move(150,400); await page.waitForTimeout(600);
  await page.evaluate(()=>document.activeElement?.blur?.());
}
const peek = async page => await page.evaluate(()=>document.querySelector("[data-sidebar-peek]")?.dataset.sidebarPeek);
const denies = async page => await page.locator(".approval-deny").count();

// Scenario: the peek sidebar is previewed AND a pending approval is on screen. One Escape.
const p = await b.newPage({viewport:{width:1440,height:1000},deviceScaleFactor:2,colorScheme:"dark"});
await openApp(p, "http://localhost:1420/?rps=0&approvals=manual");
await p.waitForTimeout(2000);
await peekWithApproval(p);
say(`peek-before=${await peek(p)} deny-rows-before=${await denies(p)}`);
await p.keyboard.press("Escape");
await p.waitForTimeout(700);
say(`peek-after=${await peek(p)} deny-rows-after=${await denies(p)}`);
await p.screenshot({path:`${out}/escape-collision.png`});

// Control: no approval anywhere. Escape must still close the peek.
const q = await b.newPage({viewport:{width:1440,height:1000},deviceScaleFactor:2,colorScheme:"dark"});
await openApp(q, "http://localhost:1420/?rps=0");
await q.waitForTimeout(2000);
await peekWithApproval(q);
say(`control answerable=${await q.locator('[data-thread-approval-surface][data-decision="pending"]:not([data-hotkeys-disabled])').count()} peek-before=${await peek(q)}`);
await q.keyboard.press("Escape");
await q.waitForTimeout(700);
say(`control peek-after=${await peek(q)}`);
await q.screenshot({path:`${out}/escape-control.png`});
writeFileSync(`${out}/escape.txt`, log.join("\n")+"\n");
await b.close();
